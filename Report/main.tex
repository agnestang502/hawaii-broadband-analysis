\begin{frame}{Defining Broadband}
    \begin{block}{The BEAD (Broadband Equity, Access, and Deployment) Program}
        \small A program to get all Americans online by funding partnerships between states or territories, communities, and stakeholders to build infrastructure where we need it and increase adoption of high-speed Internet~\cite{broadbandusa2022_bead}.
    \end{block}
    \begin{block}{Broadband}
        \small A wide bandwidth which can transport multiple signals over a ``broad'' range of frequencies and support different internet traffic types, allowing multiple data streams to be sent at once~\cite{broadbandusa2022}.
    \end{block}
    \vspace{0.3em}
    \begin{itemize}
        \item 24 million Americans do not have access to terrestrial broadband service.
        \item Expanding connectivity is linked to economic benefits.
    \end{itemize}
\end{frame}



\begin{frame}{The Rural Broadband Cost Problem}
\begin{itemize}
    \item Wired vs. Wireless Infrastructure
    \item Fiber is prioritized, but deployment costs are dependent on:
    \begin{itemize}
        \item Distance from existing infrastructure
        \item Difficult terrain
        \item Low subscriber density
    \end{itemize}
    \item Remote areas create an economic boundary where alternatives may become favorable.
\end{itemize}
\end{frame}



\begin{frame}{Research Question}
    \begin{itemize}
        \item How do geographic and cost uncertainties affect the economic threshold for fiber deployment in Hawaii?
        \item Hawaii exhibits significant terrain and population density variation.
        \item Communities are geographically dispersed across island environments.
    \end{itemize}
\end{frame}

\begin{frame}{Fiber Cost Model}
\small
\textbf{Capital Expenditure (CAPEX) per Household}
\[
\text{CAPEX}=
\frac{
\text{Distance} \times \text{ConstructionCost} \times \text{Terrain} \times \text{Deployment}
}{
\text{HouseholdsServed}
}
+ \text{DropCost}
\]
\textbf{Lifecycle Cost per Household}
\[
\text{Lifecycle Cost}
=
\text{CAPEX}
\times
\left(1+\text{Maintenance Rate}\times\text{Years}\right)
\]

\begin{itemize}
    \item Key cost drivers: Distance, terrain multiplier, deployment multiplier, subscriber drop cost, and 10-year maintenance.
\end{itemize}
\end{frame}

\begin{frame}{Discussion and Limitations}
\begin{itemize}
    \item Results rely on publicly available baseline assumptions.
    \item Representative clusters simplify fine-grained local geographic variation.
    \item Model does not attempt to replicate Hawaii's proprietary state allocation model.
    \item Provides an independent, reproducible framework for robustness evaluation.
\end{itemize}
\end{frame}


\begin{frame}{Conclusion}
\begin{itemize}
    \item Developed a publicly reproducible geospatial broadband cost model.
    \item Identified clear economic boundaries between fiber and satellite alternatives.
    \item Evaluated the robustness of technology allocation under uncertainty.
\end{itemize}
\end{frame}
